CREATE VIEW personal_account.certificates AS
 SELECT c.id,
    c.inserted,
    c.organizations_id,
    c.users_id,
    c.request_data,
    c.status,
    c.certificate,
    c.certificate_serial_number,
    c.certificate_date_from,
    c.certificate_date_to,
    c.is_active,
    c.request_id,
    c.legal_person_id,
    (unnest(xpath('certificate-request/@name'::text, c.request_data)))::character varying(100) AS cert_owner,
    o.short_name AS org_short_name
   FROM (personal_account.te_user_certificates c
     JOIN personal_account.td_organizations o ON ((c.organizations_id = o.id)));
